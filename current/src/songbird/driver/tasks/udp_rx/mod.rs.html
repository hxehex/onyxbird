53    async fn run(&mut self, interconnect: &mut Interconnect) {
54        let mut cleanup_time = Instant::now();
55        let mut playout_time = Instant::now() + TIMESTEP_LENGTH;
56        let mut byte_dest: Option<BytesMut> = None;
57
58        loop {
59            if byte_dest.is_none() {
60                byte_dest = Some(BytesMut::zeroed(VOICE_PACKET_MAX));
61            }
62
63            select! {
64                Ok((len, _addr)) = self.udp_socket.recv_from(byte_dest.as_mut().unwrap()) => {
65                    let mut pkt = byte_dest.take().unwrap();
66                    pkt.truncate(len);
67
68                    self.process_udp_message(interconnect, pkt).await;
69                },
70                msg = self.rx.recv_async() => {
71                    match msg {
72                        Ok(UdpRxMessage::ReplaceInterconnect(i)) => {
73                            *interconnect = i;
74                        },
75                        Ok(UdpRxMessage::SetConfig(new_config)) => {
76                            if let DecodeMode::Decode(old_config) = &mut self.config.decode_mode {
77                                if *old_config != new_config {
78                                    *old_config = new_config;
79                                    self.decoder_map.values_mut().for_each(|v| v.reconfigure_decoder(new_config));
80                                }
81                            }
82                        },
83                        Err(flume::RecvError::Disconnected) => break,
84                    }
85                },
86                () = tokio::time::sleep_until(playout_time) => {
87                    let mut tick = VoiceTick {
88                        speaking: HashMap::new(),
89                        silent: HashSet::new(),
90                    };
91
92                    for (ssrc, state) in &mut self.decoder_map {
93                        match state.get_voice_tick(&self.config) {
94                            Ok(Some(data)) => {
95                                tick.speaking.insert(*ssrc, data);
96                            },
97                            Ok(None) => {
98                                if !state.disconnected {
99                                    tick.silent.insert(*ssrc);
100                                }
101                            },
102                            Err(e) => {
103                                warn!("Decode error for SSRC {ssrc}: {e:?}");
104                                tick.silent.insert(*ssrc);
105                            },
106                        }
107                    }
108
109                    playout_time += TIMESTEP_LENGTH;
110
111                    drop(interconnect.events.send(EventMessage::FireCoreEvent(CoreContext::VoiceTick(tick))));
112                },
113                () = tokio::time::sleep_until(cleanup_time) => {
114                    // periodic cleanup.
115                    let now = Instant::now();
116
117                    // check ssrc map to see if the WS task has informed us of any disconnects.
118                    loop {
119                        // This is structured in an odd way to prevent deadlocks.
120                        // while-let seemed to keep the dashmap iter() alive for block scope, rather than
121                        // just the initialiser.
122                        let id = {
123                            if let Some(id) = self.ssrc_signalling.disconnected_users.iter().next().map(|v| *v.key()) {
124                                id
125                            } else {
126                                break;
127                            }
128                        };
129
130                        _ = self.ssrc_signalling.disconnected_users.remove(&id);
131                        if let Some((_, ssrc)) = self.ssrc_signalling.user_ssrc_map.remove(&id) {
132                            let _ = self.ssrc_signalling.ssrc_user_map.remove(&ssrc);
133
134                            if let Some(state) = self.decoder_map.get_mut(&ssrc) {
135                                // don't cleanup immediately: leave for later cycle
136                                // this is key with reorder/jitter buffers where we may
137                                // still need to decode post disconnect for ~0.2s.
138                                state.prune_time = now + Duration::from_secs(1);
139                                state.disconnected = true;
140                            }
141                        }
142                    }
143
144                    // now remove all dead ssrcs.
145                    self.decoder_map.retain(|_, v| v.prune_time > now);
146
147                    cleanup_time = now + Duration::from_secs(5);
148                },
149            }
150        }
151    }

153    async fn process_udp_message(&mut self, interconnect: &Interconnect, mut packet: BytesMut) {
154        // NOTE: errors here (and in general for UDP) are not fatal to the connection.
155        // Panics should be avoided due to adversarial nature of rx'd packets,
156        // but correct handling should not prompt a reconnect.
157        //
158        // For simplicity, if the event task fails then we nominate the mixing thread
159        // to rebuild their context etc. (hence, the `let _ =` statements.), as it will
160        // try to make contact every 20ms.
161        let crypto_mode = self.crypto_mode;
162
163        match demux::demux_mut(packet.as_mut()) {
164            DemuxedMut::Rtp(mut rtp) => {
165                if !rtp_valid(&rtp.to_immutable()) {
166                    return;
167                }
168
169                let ssrc = rtp.get_ssrc();
170                let has_extension = rtp.get_extension() != 0;
171
172                let mut packet_data = if self.config.decode_mode.should_decrypt() {
173                    let out = self.cipher.decrypt_rtp_in_place(&mut rtp).map(|(s, t)| {
174                        if rtp.get_padding() != 0 {
175                            let payload = rtp.payload();
176                            let padding_count = payload[payload.len() - t - 1] as usize;
177                            (s, t + padding_count, true)
178                        } else {
179                            (s, t, true)
180                        }
181                    });
182
183                    if let Err(ref e) = out { warn!("RTP decryption failed: {:?}", e); }
184                    out.ok()
185                } else {
186                    None
187                };
188
189                let mut shrinkage = 0;
190                let mut should_drop = false;
191
192                if let Some((rtp_body_start, rtp_body_tail, decrypted)) = packet_data {
193                    let payload = rtp.payload_mut();
194                    let payload_length = payload.len();
195                    
196                    // FIX: Read extension length from index 2 and 3 of payload (the actual RTP extension header)
197                    let mut ext_len = 0;
198                    if has_extension && payload.len() >= 4 {
199                        let words = u16::from_be_bytes([payload[2], payload[3]]) as usize;
200                        ext_len = 4 + (words * 4);
201                    }
202
203                    // DAVE Ciphertext starts immediately after the full extension (header + payload)
204                    let cipher_start = ext_len;
205                    let cipher_end = payload_length.saturating_sub(rtp_body_tail);
206
207                    let dave_version = self.dave_protocol_version.load(Ordering::Relaxed);
208
209                    if cipher_start < cipher_end {
210                        let body = &mut payload[cipher_start..cipher_end];
211                        let body_length = body.len();
212
213                        let has_marker = body_length >= 11 && body[body_length - DAVE_MAGIC_MARKER.len()..] == DAVE_MAGIC_MARKER[..];
214
215                        if decrypted && dave_version != 0 {
216                            if has_marker {
217                                let mut decrypted_successfully = false;
218                                
219                                if let Some(user_id) = self.ssrc_signalling.ssrc_user_map.get(&ssrc) {
220                                    if let Some(ref mut dave_session) = *self.dave_session.write().unwrap() {
221                                        if dave_session.is_ready() {
222                                            match dave_session.decrypt(user_id.0, davey::MediaType::AUDIO, body) {
223                                                Ok(decrypted_body) => {
224                                                    shrinkage = body.len() - decrypted_body.len();
225                                                    body[..decrypted_body.len()].copy_from_slice(&decrypted_body);
226                                                    decrypted_successfully = true;
227                                                },
228                                                Err(_) => {}
229                                            }
230                                        }
231                                    }
232                                }
233                                
234                                if !decrypted_successfully {
235                                    should_drop = true;
236                                }
237                            }
238                        }
239                    }
240                    
241                    if shrinkage > 0 {
242                        // Shift the Transport MAC left to cover the gap left by the stripped DAVE MAC
243                        let suffix_start = payload_length - rtp_body_tail;
244                        let suffix_end = payload_length;
245                        payload.copy_within(suffix_start..suffix_end, suffix_start - shrinkage);
246                    }
247                    
248                    packet_data = Some((rtp_body_start, rtp_body_tail, decrypted));
249                }
250
251                if should_drop {
252                    return; 
253                }
254
255                let (rtp_body_start, rtp_body_tail, decrypted) = packet_data.unwrap_or_else(|| {
256                    (
257                        crypto_mode.payload_prefix_len(),
258                        crypto_mode.payload_suffix_len(),
259                        false,
260                    )
261                });
262
263                drop(rtp);
264
265                if shrinkage > 0 {
266                    let new_len = packet.len() - shrinkage;
267                    packet.truncate(new_len);
268                }
269
270                let rtp_immutable = RtpPacket::new(&packet).unwrap();
271
272                let entry = self
273                    .decoder_map
274                    .entry(ssrc)
275                    .or_insert_with(|| SsrcState::new(&rtp_immutable, crypto_mode, &self.config));
276
277                entry.refresh_timer(self.config.decode_state_timeout.into());
278
279                let store_pkt = StoredPacket {
280                    packet: packet.freeze(),
281                    decrypted,
282                };
283                
284                let packet_frozen = store_pkt.packet.clone();
285                entry.store_packet(store_pkt, &self.config);
286
287                drop(interconnect.events.send(EventMessage::FireCoreEvent(
288                    CoreContext::RtpPacket(InternalRtpPacket {
289                        packet: packet_frozen,
290                        payload_offset: rtp_body_start,
291                        payload_end_pad: rtp_body_tail,
292                    }),
293                )));
294            },
295            DemuxedMut::Rtcp(mut rtcp) => {
296                let packet_data = if self.config.decode_mode.should_decrypt() {
297                    let out = self.cipher.decrypt_rtcp_in_place(&mut rtcp);
298
299                    if let Err(ref e) = out {
300                        warn!("RTCP decryption failed: {:?}", e);
301                    }
302
303                    out.ok()
304                } else {
305                    None
306                };
307
308                let (start, tail) = packet_data.unwrap_or_else(|| {
309                    (
310                        crypto_mode.payload_prefix_len(),
311                        crypto_mode.payload_suffix_len(),
312                    )
313                });
314
315                drop(interconnect.events.send(EventMessage::FireCoreEvent(
316                    CoreContext::RtcpPacket(InternalRtcpPacket {
317                        packet: packet.freeze(),
318                        payload_offset: start,
319                        payload_end_pad: tail,
320                    }),
321                )));
322            },
323            DemuxedMut::FailedParse(t) => {
324                warn!("Failed to parse message of type {:?}.", t);
325            },
326            DemuxedMut::TooSmall => {
327                warn!("Illegal UDP packet from voice server.");
328            },
329        }
330    }
331}

333#[instrument(skip(interconnect, rx, cipher))]
334pub(crate) async fn runner(
335    mut interconnect: Interconnect,
336    rx: Receiver<UdpRxMessage>,
337    cipher: Cipher,
338    crypto_mode: CryptoMode,
339    config: Config,
340    udp_socket: UdpSocket,
341    ssrc_signalling: Arc<SsrcTracker>,
342    dave_session: Arc<RwLock<Option<davey::DaveSession>>>,
343    dave_protocol_version: Arc<AtomicU16>,
344) {
345    trace!("UDP receive handle started.");
346
347    let mut state = UdpRx {
348        cipher,
349        crypto_mode,
350        decoder_map: HashMap::new(),
351        config,
352        rx,
353        ssrc_signalling,
354        udp_socket,
355        dave_session,
356        dave_protocol_version,
357    };
358
359    state.run(&mut interconnect).await;
360
361    trace!("UDP receive handle stopped.");
362}
